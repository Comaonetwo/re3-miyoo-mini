cmake_minimum_required(VERSION 3.20)
project(re3_cpu_probe C CXX)
option(SOFT_PIXEL_STATS "Collect detailed per-pixel renderer diagnostics" ON)
if(SOFT_PIXEL_STATS)
 add_compile_definitions(SOFT_PIXEL_STATS=1)
else()
 add_compile_definitions(SOFT_PIXEL_STATS=0)
endif()
set(CMAKE_CXX_STANDARD 11)
if(NOT RE3_CPU_SOURCE OR NOT LIBRW_SOURCE)
  message(FATAL_ERROR "Provide RE3_CPU_SOURCE (prepare-re3-cpu.py output) and pinned LIBRW_SOURCE")
endif()
set(EXPERIMENT "${CMAKE_CURRENT_LIST_DIR}/../..")
set(RE3_SOURCE "${RE3_CPU_SOURCE}")
set(SOFT_SDL OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${EXPERIMENT}" experiment)
file(GLOB_RECURSE CORE_SOURCES "${RE3_CPU_SOURCE}/src/*.cpp")
list(FILTER CORE_SOURCES EXCLUDE REGEX "/skel/(glfw|win)/|/audio/eax/")
file(GLOB_RECURSE CORE_HEADERS "${RE3_CPU_SOURCE}/src/*.h")
set(CORE_DIRS "${RE3_CPU_SOURCE}" "${EXPERIMENT}/src")
foreach(H ${CORE_HEADERS})
 get_filename_component(D "${H}" DIRECTORY)
 list(APPEND CORE_DIRS "${D}")
endforeach()
list(REMOVE_DUPLICATES CORE_DIRS)
add_executable(re3_cpu ${CORE_SOURCES} "${EXPERIMENT}/src/re3_cpu_particles.cpp" "${EXPERIMENT}/src/re3_cpu_world_lod.cpp" "${EXPERIMENT}/src/re3_cpu_draw_order.cpp" "${EXPERIMENT}/src/re3_cpu_detail.cpp" "${EXPERIMENT}/src/re3_cpu_world_fade.cpp" "${EXPERIMENT}/src/re3_audio_host.cpp" "${EXPERIMENT}/src/re3_cpu_platform.cpp" "${EXPERIMENT}/src/re3_cpu_boot.cpp" "${EXPERIMENT}/src/re3_cpu_garage.cpp" "${EXPERIMENT}/src/re3_cpu_console.cpp" "${EXPERIMENT}/src/fb_blit.cpp")
target_include_directories(re3_cpu PRIVATE ${CORE_DIRS})
target_compile_definitions(re3_cpu PRIVATE LIBRW RW_NULL MIYOO_CPU CMAKE_NO_AUTOLINK NDEBUG)
target_compile_options(re3_cpu PRIVATE -w -include "${EXPERIMENT}/src/re3_soft_compat.hpp")
add_library(cpu_charset "${RE3_CPU_SOURCE}/cpu_charset.cpp")
target_include_directories(cpu_charset PRIVATE "${EXPERIMENT}/src" "${LIBRW_SOURCE}/src")
target_compile_definitions(cpu_charset PRIVATE MIYOO_CPU RW_NULL)
target_link_libraries(cpu_charset PUBLIC librw::librw)
target_link_libraries(re3_cpu PRIVATE soft_device cpu_charset)
if(SOFT_SANITIZERS)
 target_compile_options(re3_cpu PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(re3_cpu PRIVATE -fsanitize=address,undefined)
endif()

add_library(cpu_native_texture "${EXPERIMENT}/src/re3_cpu_texture.cpp")
target_include_directories(cpu_native_texture PRIVATE ${CORE_DIRS})
target_compile_definitions(cpu_native_texture PRIVATE LIBRW RW_NULL MIYOO_CPU NDEBUG)
target_compile_options(cpu_native_texture PRIVATE -include "${EXPERIMENT}/src/re3_soft_compat.hpp")
target_link_libraries(cpu_native_texture PUBLIC soft_device)
target_link_libraries(re3_cpu PRIVATE cpu_native_texture)
if(NOT CMAKE_CROSSCOMPILING)
 enable_testing()
 add_executable(re3_texture_tests "${EXPERIMENT}/tests/re3_texture_tests.cpp")
 target_link_libraries(re3_texture_tests PRIVATE cpu_native_texture)
 target_compile_options(re3_texture_tests PRIVATE -UNDEBUG)
 add_test(NAME native_txd COMMAND re3_texture_tests)
endif()
if(SOFT_SANITIZERS)
 target_compile_options(cpu_native_texture PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()

if(SOFT_SANITIZERS)
 target_compile_options(cpu_charset PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 if(TARGET re3_texture_tests)
  target_compile_options(re3_texture_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(re3_texture_tests PRIVATE -fsanitize=address,undefined)
 endif()
endif()

if(NOT CMAKE_CROSSCOMPILING)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 find_package(Threads REQUIRED)
 add_executable(re3_audio_test_helper "${EXPERIMENT}/src/re3_audio_helper.c")
 target_link_libraries(re3_audio_test_helper PRIVATE ${CMAKE_DL_LIBS})
 add_library(audio_test_sdl SHARED "${EXPERIMENT}/tests/audio_sdl_stub.c")
 target_link_libraries(audio_test_sdl PRIVATE Threads::Threads)
 add_test(NAME audio_recovery COMMAND ${Python3_EXECUTABLE} "${EXPERIMENT}/tests/audio_recovery_test.py" $<TARGET_FILE:re3_audio_test_helper> $<TARGET_FILE:audio_test_sdl>)
 add_test(NAME streamed_audio COMMAND ${Python3_EXECUTABLE} "${EXPERIMENT}/tests/audio_ipc_test.py" $<TARGET_FILE:re3_audio_test_helper> $<TARGET_FILE:audio_test_sdl>)
 if(SOFT_SANITIZERS)
  foreach(T re3_audio_test_helper audio_test_sdl)
   target_compile_options(${T} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
   target_link_options(${T} PRIVATE -fsanitize=address,undefined)
  endforeach()
 endif()
endif()
